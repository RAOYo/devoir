Algorithme : somme de 1 à n

Variables: i: entier;
           n,somme: réels;
Début    
    AFFICHER("Veuillez entrer le nombre n:");
    SAISIR(n);
    AFFICHER("0");
    
    somme=0;
    Pour(i=1 ; i<=n ; i++)
        somme <- somme + i;
        AFFICHER("+"i);
    finPour
    
    AFFICHER(somme);
    
Fin
