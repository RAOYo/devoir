Variables:  a,b,c,x,x0,x1,x2,x3,x4,delta :réels;

    AFFICHER("---Résolution de l'équation du second degré---\n\n");

    AFFICHER("Veuillez entrer la valeur de b:");

        		delta <- (pow(b,2) - (4 * a *c));
        		Si (delta > 0) alors 
				x1<-(-b-sqrt(delta))/(2*a);

			    	AFFICHER("les 2 solutions rélles sont=",x1,x2);

            				x0 <- -b /(2*a);
            				AFFICHER("l'unique solution est: ",x0);
        				Sinon
            					x3<- -b /(2*a);
            					x4<- (sqrt(-delta))/ (2*a);
            					AFFICHER("les solutions complexes sont:" ,x3, "-i",x4, "ET" ,x3,"+i",x4);

            	         finsi
